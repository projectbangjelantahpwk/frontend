//Android App

*Tabel Profil & Akses (profiles)*
CREATE TABLE profiles (
  id UUID REFERENCES auth.users NOT NULL PRIMARY KEY,
  id_lapangan TEXT UNIQUE NOT NULL, -- Contoh ID: 'JLC-082299'
  nama TEXT NOT NULL,
  nomor_hp TEXT, -- Bisa format '+628...' atau '08...'
  role TEXT DEFAULT 'lapangan', -- 'lapangan', 'admin', 'office'
  tipe_kendaraan TEXT, -- 'motor', 'mobil_driver', 'mobil_helper'
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

**Tabel Presensi Kehadiran (presensi)**
CREATE TABLE presensi (
  id SERIAL PRIMARY KEY,
  user_id UUID REFERENCES profiles(id),
  tanggal DATE NOT NULL DEFAULT CURRENT_DATE,
  waktu_tiba TIME,
  waktu_pulang TIME,
  status_tiba TEXT, -- 'Tepat Waktu', 'Telat'
  status_pulang TEXT, -- 'Tepat Waktu', 'Lembur', 'Nginep'
  UNIQUE(user_id, tanggal) -- Biar gak bisa dobel absen di hari yang sama
);

*Tabel Pengeluaran Tumpuk (pengeluaran & rincian_pengeluaran)*
CREATE TABLE pengeluaran (
  id SERIAL PRIMARY KEY,
  user_id UUID REFERENCES profiles(id),
  tanggal DATE NOT NULL DEFAULT CURRENT_DATE,
  total_harian DECIMAL DEFAULT 0,
  UNIQUE(user_id, tanggal)
);

CREATE TABLE rincian_pengeluaran (
  id SERIAL PRIMARY KEY,
  pengeluaran_id INT REFERENCES pengeluaran(id) ON DELETE CASCADE,
  nama_item TEXT NOT NULL,
  nominal DECIMAL NOT NULL,
  jam_input TIME DEFAULT CURRENT_TIME
);

*Tabel Operasional / Navigasi Rute (penjemputan)*
CREATE TABLE penjemputan (
  id SERIAL PRIMARY KEY,
  mitra_id INT REFERENCES mitra(id),
  driver_id UUID REFERENCES profiles(id), -- Yang pegang navigasi
  helper_id UUID REFERENCES profiles(id), -- Nullable kalau dia bawa Motor sendirian
  status TEXT DEFAULT 'Belum', -- 'Belum', 'OTW', 'Proses', 'Selesai'
  tanggal DATE DEFAULT CURRENT_DATE
);

*Tabel Timbangan & Uang (transaksi & transaksi_item)*
CREATE TABLE transaksi (
  id SERIAL PRIMARY KEY,
  penjemputan_id INT REFERENCES penjemputan(id),
  kalkulator_user_id UUID REFERENCES profiles(id), -- Helper atau Supir Motor
  metode_bayar TEXT, -- 'Cash', 'QRIS', 'Transfer'
  subtotal DECIMAL,
  potongan_dp DECIMAL DEFAULT 0,
  grand_total DECIMAL,
  is_paid_by_admin BOOLEAN DEFAULT FALSE, -- Kalau FALSE, di Profil warnanya Merah!
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE transaksi_item (
  id SERIAL PRIMARY KEY,
  transaksi_id INT REFERENCES transaksi(id) ON DELETE CASCADE,
  grade_minyak TEXT,
  volume DECIMAL,
  harga_unit DECIMAL
);

*Slip Gaji*
CREATE TABLE slip_gaji (
  id SERIAL PRIMARY KEY,
  user_id UUID REFERENCES profiles(id) NOT NULL,
  periode TEXT NOT NULL, -- Contoh: 'Minggu 3 - Mei 2026' atau 'Bulan Mei 2026'
  
  -- Komponen Pemasukan & Potongan
  dana_talang DECIMAL DEFAULT 0,
  pengeluaran_karyawan DECIMAL DEFAULT 0,
  gaji_karyawan DECIMAL DEFAULT 0,
  uang_lembur DECIMAL DEFAULT 0,
  uang_kesehatan DECIMAL DEFAULT 0,
  bonus_tepat_waktu DECIMAL DEFAULT 0,
  bonus_mingguan DECIMAL DEFAULT 0,
  
  -- Status Pembayaran (Trigger untuk warna merah di aplikasi)
  is_paid_by_admin BOOLEAN DEFAULT FALSE,
  
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  UNIQUE(user_id, periode) -- Biar admin gak dobel input di periode yang sama
);

*Mitra*
CREATE TABLE mitra (
  id SERIAL PRIMARY KEY,
  nama_mitra TEXT NOT NULL,
  alamat TEXT,
  link_gmaps TEXT, -- Menyimpan link Google Maps lokasi mitra
  nomor_hp TEXT,
  total_order INT DEFAULT 0, -- Menyimpan berapa kali order/penjemputan sukses
  total_transaksi DECIMAL DEFAULT 0, -- Menyimpan akumulasi total uang transaksi (Rp)
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

Policynya berikut :
-- 1. BERSIH-BERSIH ATURAN LAMA (Biar nggak error bentrok)
DROP POLICY IF EXISTS "Update profil sendiri" ON profiles;
DROP POLICY IF EXISTS "CRUD Presensi Pribadi" ON presensi;
DROP POLICY IF EXISTS "Lihat Gaji Sendiri" ON slip_gaji;
DROP POLICY IF EXISTS "Hanya Admin kelola Gaji" ON slip_gaji;
DROP POLICY IF EXISTS "CRUD Pengeluaran Pribadi" ON pengeluaran;
DROP POLICY IF EXISTS "CRUD Rincian Pengeluaran Pribadi" ON rincian_pengeluaran;
DROP POLICY IF EXISTS "CRUD Transaksi Sendiri" ON transaksi;
DROP POLICY IF EXISTS "CRUD Transaksi Item Sendiri" ON transaksi_item;

-- ==========================================
-- 2. POLICY PROFILES (DIKUASAI: Owner, HRD, Admin)
-- ==========================================
CREATE POLICY "Update profil sendiri dan Managemen" ON profiles 
FOR UPDATE USING (
  auth.uid() = id OR 
  (SELECT role FROM profiles WHERE id = auth.uid()) IN ('owner', 'hrd', 'admin')
);

-- ==========================================
-- 3. POLICY PRESENSI (DIKUASAI: Owner, HRD)
-- ==========================================
CREATE POLICY "CRUD Presensi Khusus" ON presensi 
FOR ALL USING (
  auth.uid() = user_id OR 
  (SELECT role FROM profiles WHERE id = auth.uid()) IN ('owner', 'hrd', 'admin') -- Admin kadang butuh lihat absen buat ngatur rute
);

-- ==========================================
-- 4. POLICY SLIP GAJI (DIKUASAI: Owner, HRD, Keuangan)
-- ==========================================
CREATE POLICY "Lihat Gaji Pribadi" ON slip_gaji 
FOR SELECT USING (
  auth.uid() = user_id OR 
  (SELECT role FROM profiles WHERE id = auth.uid()) IN ('owner', 'hrd', 'keuangan')
);

CREATE POLICY "Kelola Gaji" ON slip_gaji 
FOR ALL USING (
  (SELECT role FROM profiles WHERE id = auth.uid()) IN ('owner', 'hrd', 'keuangan')
);

-- ==========================================
-- 5. POLICY PENGELUARAN KAS JALAN (DIKUASAI: Owner, Keuangan, Admin)
-- ==========================================
CREATE POLICY "CRUD Pengeluaran Harian" ON pengeluaran 
FOR ALL USING (
  auth.uid() = user_id OR 
  (SELECT role FROM profiles WHERE id = auth.uid()) IN ('owner', 'keuangan', 'admin')
);

CREATE POLICY "CRUD Rincian Pengeluaran Harian" ON rincian_pengeluaran 
FOR ALL USING (
  EXISTS (SELECT 1 FROM pengeluaran WHERE id = rincian_pengeluaran.pengeluaran_id AND user_id = auth.uid()) 
  OR (SELECT role FROM profiles WHERE id = auth.uid()) IN ('owner', 'keuangan', 'admin')
);

-- ==========================================
-- 6. POLICY TRANSAKSI UANG / TIMBANGAN (DIKUASAI: Owner, Keuangan, Admin)
-- ==========================================
CREATE POLICY "CRUD Transaksi Penjemputan" ON transaksi 
FOR ALL USING (
  auth.uid() = kalkulator_user_id OR 
  (SELECT role FROM profiles WHERE id = auth.uid()) IN ('owner', 'keuangan', 'admin')
);

CREATE POLICY "CRUD Transaksi Item Penjemputan" ON transaksi_item 
FOR ALL USING (
  EXISTS (SELECT 1 FROM transaksi WHERE id = transaksi_item.transaksi_id AND kalkulator_user_id = auth.uid())
  OR (SELECT role FROM profiles WHERE id = auth.uid()) IN ('owner', 'keuangan', 'admin')
);
